// Define Pins
const int trigPin = 9; // Trig pin connected to Arduino Pin 9
const int echoPin = 10; // Echo pin connected to Arduino Pin 10

// Variables
long duration; // Stores the time it takes for the sound wave to travel
int distance;  // Stores the calculated distance

void setup() {
  Serial.begin(9600); // Start serial communication
  pinMode(trigPin, OUTPUT); // Set trigPin as an OUTPUT
  pinMode(echoPin, INPUT);  // Set echoPin as an INPUT
  Serial.println("Ultrasonic Sensor Test");
}

void loop() {
  // 1. Clear the trigPin by setting it LOW for a short duration
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);

  // 2. Send a 10 microsecond pulse to trigger the sensor
  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);

  // 3. Measure the duration of the echo pulse (time from trigger to echo)
  // pulseIn(pin, value) returns the time in microseconds
  duration = pulseIn(echoPin, HIGH);

  // 4. Calculate the distance in centimeters
  // Distance = (Time * Speed of Sound) / 2
  // Speed of sound in air is ~340 m/s or ~0.034 cm/µs
  distance = duration * 0.034 / 2;

  // 5. Print the distance to the Serial Monitor
  Serial.print("Distance: ");
  Serial.print(distance);
  Serial.println(" cm");

  delay(1000); // Wait for a second before the next reading
}
